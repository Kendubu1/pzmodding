module Base
{
    /*
        The Fate Token. Carrying one when you die spends it: Permadeath Lock
        records the death but does not lock you out, and your next character
        collects the skills the dead one had.

        Build 42 item syntax - ItemType replaces Build 41's Type.

        DisplayCategory must be a stock Build 42 category. One the game has no
        name for renders in the inventory as its own translation key -
        "IGUI_ItemCat_..." - and supplying that name from a mod does not
        reliably fix it. "Junk" is where Build 42 files oddments, needs no label
        from us, and reads correctly. Note that Build 41's "Misc" is gone.
    */
    item FateToken
    {
        DisplayCategory = Junk,
        ItemType = base:normal,
        DisplayName = Fate Token,
        Icon = FateToken,
        Weight = 0.1,
        Tooltip = Tooltip_FateToken,
    }
}
