module Base
{
    /*
        The Fate Token. Carrying one when you die spends it: Permadeath Lock
        records the death but does not lock you out, and your next character
        collects the skills the dead one had.

        Build 42 item syntax - ItemType replaces Build 41's Type.

        DisplayCategory is the mod's own category, not a vanilla one, and its
        label ships as IGUI_ItemCat_Fate in Translate/EN/IG_UI_EN.txt. Note the
        file name: the game does not read a file called IGUI_EN.txt, so a label
        put there shows as its own key. Naming a vanilla category that Build 42
        has since dropped fails the same way - "Misc" rendered as
        IGUI_ItemCat_Misc.
    */
    item FateToken
    {
        DisplayCategory = Fate,
        ItemType = base:normal,
        DisplayName = Fate Token,
        Icon = FateToken,
        Weight = 0.1,
        Tooltip = Tooltip_FateToken,
    }
}
